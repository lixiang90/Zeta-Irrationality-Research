\[

\[
 F_{n,r}(z)=\Lambda_n z^{16n}(z-1)^{3n}I_z(h,j,k,l,m)
 =P_{n,r}(z)-Q_{n,r}(z)\operatorname{Li}_2(1/z),

\[
